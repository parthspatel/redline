fn sc1_redline_diff_returns_complete_result() {

        "The quick brown fox jumps over the lazy dog.",

    // Verify metrics are present
    assert!(!result.metrics.is_empty(), "Metrics should be computed");
    assert!(result.metrics.contains_key("word_count"), "Should have word_count metric");
    assert!(result.metrics.contains_key("similarity_ratio"), "Should have similarity_ratio metric");

    // Verify analysis is present (Comprehensive default)
    assert!(result.analysis.is_some(), "Analysis should be present for Comprehensive preset");
    let report = result.analysis.as_ref().unwrap();

}

#[test]
fn sc3_filter_replace_operations_by_length() {
    let config = ConfigBuilder::new().build().unwrap();
    let redline = Redline::new(config).unwrap();

    // Create texts where some replaces are long (>10 tokens) and some are short
    let source = "word1 word2 word3 word4 word5 word6 word7 word8 word9 word10 word11 word12 end";
    let target = "changed1 changed2 changed3 changed4 changed5 changed6 changed7 changed8 changed9 changed10 changed11 changed12 end";

    let result = redline.diff(source, target).unwrap();

    // Filter: Replace AND min_length > 10 (token count)
    let filter = Filter::kind(EditKind::Replace) & Filter::min_length(10);
    let filtered: Vec<_> = result.filter(&filter).collect();

    // All filtered operations should be Replace with length >= 10
    for op in &filtered {
        assert_eq!(op.kind, EditKind::Replace);
        assert!(op.source_len().max(op.target_len()) >= 10,
            "Operation should have length >= 10, got source_len={} target_len={}",
            op.source_len(), op.target_len());
    }

    // Also verify the filter EXCLUDED short replaces (if any)
    let all_replaces: Vec<_> = result.filter(&Filter::kind(EditKind::Replace)).collect();
    // filtered should be a subset of all_replaces
    assert!(filtered.len() <= all_replaces.len());
}

```rust
#[test]
fn sc4_memory_usage_10k_words_under_50mb() {
    // Generate a 10K word document pair
    let words: Vec<&str> = vec![
        "the", "quick", "brown", "fox", "jumps", "over", "lazy", "dog",
        "alpha", "bravo", "charlie", "delta", "echo", "foxtrot", "golf",
        "hotel", "india", "juliet", "kilo", "lima", "mike", "november",
        "oscar", "papa", "quebec", "romeo", "sierra", "tango", "uniform",
        "victor", "whiskey", "xray", "yankee", "zulu", "one", "two",
        "three", "four", "five", "six", "seven", "eight", "nine", "ten",
        "this", "that", "with", "from", "into", "upon",
    ];

    // Build source: 10K words
    let source: String = (0..10_000)
        .map(|i| words[i % words.len()])
        .collect::<Vec<_>>()
        .join(" ");

    // Build target: ~10K words with ~10% changes
    let target: String = (0..10_000)
        .map(|i| {
            if i % 10 == 0 {
                // Every 10th word is different
                words[(i + 7) % words.len()]
            } else {
                words[i % words.len()]
            }
        })
        .collect::<Vec<_>>()
        .join(" ");

    // Measure memory before
    let mem_before = get_process_memory_bytes();

    // Run the full pipeline
    let config = ConfigBuilder::new().build().unwrap();
    let redline = Redline::new(config).unwrap();
    let result = redline.diff(&source, &target).unwrap();

    // Measure memory after
    let mem_after = get_process_memory_bytes();

    let mem_used_bytes = mem_after.saturating_sub(mem_before);
    let mem_used_mb = mem_used_bytes as f64 / (1024.0 * 1024.0);

    // Verify result is complete
    assert!(!result.diff.operations.is_empty());
    assert!(!result.metrics.is_empty());
    assert!(result.analysis.is_some());

    // Verify memory is under 50MB
    // Note: Process memory measurement is approximate.
    // Use a generous bound — the requirement is <50MB.
    assert!(
        mem_used_mb < 50.0,
        "Memory usage should be <50MB, measured approximately {:.1}MB",
        mem_used_mb
    );

    // Print for informational purposes
    eprintln!(
        "10K word document pair analysis: ~{:.1}MB (limit: 50MB)",
        mem_used_mb
    );
}

/// Get approximate process memory usage in bytes.
/// Uses platform-specific methods.
fn get_process_memory_bytes() -> usize {
    // On macOS, use mach APIs or proc info
    // On Linux, read /proc/self/statm
    // Cross-platform fallback: use a simpler approach

    #[cfg(target_os = "macos")]
    {
        use std::process::Command;
        let output = Command::new("ps")
            .args(["-o", "rss=", "-p", &std::process::id().to_string()])
            .output()
            .ok();
        if let Some(output) = output {
            let rss_kb = String::from_utf8_lossy(&output.stdout)
                .trim()
                .parse::<usize>()
                .unwrap_or(0);
            return rss_kb * 1024; // Convert KB to bytes
        }
        0
    }

    #[cfg(target_os = "linux")]
    {
        if let Ok(statm) = std::fs::read_to_string("/proc/self/statm") {
            let rss_pages = statm.split_whitespace()
                .nth(1)
                .and_then(|s| s.parse::<usize>().ok())
                .unwrap_or(0);
            return rss_pages * 4096; // Assume 4K pages
        }
        0
    }

    #[cfg(not(any(target_os = "macos", target_os = "linux")))]
    {
        0 // Can't measure on this platform
    }
}
```

                    assert!(result.is_ok(), "diff() should not fail: {:?}", result.err());

            })
        })
        .collect();
