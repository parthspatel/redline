use crate::error::StoreError;

// --- Minimal stubs (will be replaced in GREEN phase) ---

/// Opaque handle to an interned string. Copy-cheap, 4 bytes.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct StringId(pub u32);

pub struct TextStoreBuilder;  // stub
pub struct TextStore;          // stub

// --- Tests define the API contract ---
#[cfg(test)]
mod tests {
    use super::*;
    use static_assertions::assert_impl_all;
    use std::sync::Arc;

    assert_impl_all!(StringId: Copy, Clone, core::fmt::Debug, PartialEq, Eq, core::hash::Hash, Send, Sync);

    // --- StringId tests ---
    #[test]
    fn string_id_as_u32() {
        let id = StringId(42);
        assert_eq!(id.as_u32(), 42);
    }

    // --- Builder: basic interning ---
    #[test]
    fn intern_and_resolve() {
        let mut builder = TextStoreBuilder::new();
        let id = builder.intern("hello");
        assert_eq!(builder.resolve(id).unwrap(), "hello");
    }

    // --- Builder: deduplication ---
    #[test]
    fn deduplication() {
        let mut builder = TextStoreBuilder::new();
        let id1 = builder.intern("hello");
        let id2 = builder.intern("hello");
        assert_eq!(id1, id2);
        assert_eq!(builder.len(), 1);
    }

    #[test]
    fn different_strings_get_different_ids() {
        let mut builder = TextStoreBuilder::new();
        let id1 = builder.intern("hello");
        let id2 = builder.intern("world");
        assert_ne!(id1, id2);
        assert_eq!(builder.len(), 2);
    }

    // --- Promotion to Arc<TextStore> ---
    #[test]
    fn promote_to_arc_and_resolve() {
        let mut builder = TextStoreBuilder::new();
        let id1 = builder.intern("hello");
        let id2 = builder.intern("world");
        let store: Arc<TextStore> = builder.build();
        assert_eq!(store.resolve(id1).unwrap(), "hello");
        assert_eq!(store.resolve(id2).unwrap(), "world");
    }

    // --- SUCCESS CRITERION 2: 10K strings ---
    #[test]
    fn intern_10k_strings() {
        let mut builder = TextStoreBuilder::new();
        let ids: Vec<StringId> = (0..10_000)
            .map(|i| builder.intern(&format!("string_{i}")))
            .collect();

        let store = builder.build();

        for (i, id) in ids.iter().enumerate() {
            assert_eq!(store.resolve(*id).unwrap(), format!("string_{i}"));
        }
    }

    // --- Error handling ---
    #[test]
    fn id_not_found_returns_error() {
        let builder = TextStoreBuilder::new();
        let store = builder.build();
        assert!(matches!(store.resolve(StringId(999)), Err(StoreError::IdNotFound(999))));
    }

    // --- Edge cases ---
    #[test]
    fn empty_string_interning() {
        let mut builder = TextStoreBuilder::new();
        let id = builder.intern("");
        let store = builder.build();
        assert_eq!(store.resolve(id).unwrap(), "");
    }

    #[test]
    fn unicode_interning() {
        let mut builder = TextStoreBuilder::new();
        let id = builder.intern("こんにちは世界");
        let store = builder.build();
        assert_eq!(store.resolve(id).unwrap(), "こんにちは世界");
    }

    #[test]
    fn arc_is_shared() {
        let mut builder = TextStoreBuilder::new();
        let id = builder.intern("shared");
        let store = builder.build();
        let store2 = Arc::clone(&store);
        assert_eq!(store.resolve(id).unwrap(), store2.resolve(id).unwrap());
    }

    #[test]
    fn builder_is_empty_initially() {
        let builder = TextStoreBuilder::new();
        assert!(builder.is_empty());
        assert_eq!(builder.len(), 0);
    }

    #[test]
    fn store_len_matches_unique_strings() {
        let mut builder = TextStoreBuilder::new();
        builder.intern("a");
        builder.intern("b");
        builder.intern("a"); // duplicate
        let store = builder.build();
        assert_eq!(store.len(), 2);
    }
}

Replace stubs with full implementation:

**StringId:**
- Add method: as_u32(self) -> u32

**TextStoreBuilder:**
Structure (two-phase pattern from research):
- strings: Vec<&'static str> -- indexed by StringId.0
- dedup: HashMap<&'static str, StringId> -- hashbrown
- arena: Bump -- declared LAST (dropped last)

Methods:
- new() -> Self
- intern(&mut self, s: &str) -> StringId: check dedup → allocate in arena → transmute to 'static → push to strings, insert dedup
- resolve(&self, id: StringId) -> Result<&str, StoreError>
- len(), is_empty()

**TextStore:**
Structure:
- buffer: Vec<u8> -- contiguous string bytes
- offsets: Vec<(u32, u32)> -- (offset, len)

Methods:
- resolve(&self, id: StringId) -> Result<&str, StoreError>
- len(), is_empty()

**TextStoreBuilder::build(self) -> Arc<TextStore>:**
- Iterate strings, copy bytes to buffer, record offsets
- No arena refs in final TextStore (Miri-clean)

For no_std: use alloc::sync::Arc, alloc::vec::Vec when not(feature = "std").
