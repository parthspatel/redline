```rust
use crate::span::Span;
use crate::text_store::StringId;

// Minimal stubs
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
#[repr(u8)]
pub enum TokenKind {
    Regular = 0,
    Unknown = 1,
    Special = 2,
    Continuation = 3,
}

pub struct Token;  // stub — will fail tests

#[cfg(test)]
mod tests {
    use super::*;
    use static_assertions::assert_impl_all;

    // SUCCESS CRITERION 1: This line must compile
    assert_impl_all!(Token: Copy, Clone, core::fmt::Debug, PartialEq, Send, Sync);
    assert_impl_all!(TokenKind: Copy, Clone, core::fmt::Debug, PartialEq, Eq, core::hash::Hash);

    // Size constraint
    const _: () = assert!(core::mem::size_of::<Token>() <= 24);

    #[test]
    fn token_creation_and_accessors() {
        let token = Token::new(StringId(0), Span::new(0, 5), TokenKind::Regular);
        assert_eq!(token.text_id, StringId(0));
        assert_eq!(token.span, Span::new(0, 5));
        assert_eq!(token.kind, TokenKind::Regular);
    }

    #[test]
    fn token_is_special() {
        let token = Token::new(StringId(1), Span::new(0, 0), TokenKind::Special);
        assert!(token.is_special());
        assert!(!token.is_continuation());
        assert!(!token.is_unknown());
    }

    #[test]
    fn token_is_continuation() {
        let token = Token::new(StringId(2), Span::new(5, 10), TokenKind::Continuation);
        assert!(token.is_continuation());
        assert!(!token.is_special());
    }

    #[test]
    fn token_is_unknown() {
        let token = Token::new(StringId(3), Span::new(0, 3), TokenKind::Unknown);
        assert!(token.is_unknown());
    }

    #[test]
    fn token_kind_repr() {
        assert_eq!(TokenKind::Regular as u8, 0);
        assert_eq!(TokenKind::Unknown as u8, 1);
        assert_eq!(TokenKind::Special as u8, 2);
        assert_eq!(TokenKind::Continuation as u8, 3);
    }

    #[test]
    fn token_equality() {
        let a = Token::new(StringId(0), Span::new(0, 5), TokenKind::Regular);
        let b = Token::new(StringId(0), Span::new(0, 5), TokenKind::Regular);
        let c = Token::new(StringId(1), Span::new(0, 5), TokenKind::Regular);
        assert_eq!(a, b);
        assert_ne!(a, c);
    }

    #[test]
    fn token_is_copy() {
        let a = Token::new(StringId(0), Span::new(0, 5), TokenKind::Regular);
        let b = a; // Copy
        let c = a; // Still valid because Copy
        assert_eq!(b, c);
    }
}
```

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
pub struct Token {
    pub text_id: StringId,
    pub span: Span,
    pub kind: TokenKind,
}

impl Token {
    pub fn new(text_id: StringId, span: Span, kind: TokenKind) -> Self {
        Self { text_id, span, kind }
    }
    pub fn is_special(&self) -> bool { matches!(self.kind, TokenKind::Special) }
    pub fn is_continuation(&self) -> bool { matches!(self.kind, TokenKind::Continuation) }
    pub fn is_unknown(&self) -> bool { matches!(self.kind, TokenKind::Unknown) }
}
```
