use crate::error::NormalizeError;

// Minimal stub
pub struct CharMapping {
    alignments: Vec<(u32, u32)>,
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn identity_round_trips() {
        let m = CharMapping::identity(10);
        for i in 0..10 {
            assert_eq!(m.to_normalized(i).unwrap(), i);
            assert_eq!(m.to_original(i).unwrap(), i);
        }
    }

    #[test]
    fn simple_offset_mapping() {
        // Original "ab" -> Normalized "a_b" (insertion shifts b from pos 1 to pos 2)
        let m = CharMapping::new(vec![(0, 0), (1, 2)]);
        assert_eq!(m.to_normalized(0).unwrap(), 0);
        assert_eq!(m.to_normalized(1).unwrap(), 2);
        assert_eq!(m.to_original(0).unwrap(), 0);
        assert_eq!(m.to_original(2).unwrap(), 1);
    }

    #[test]
    fn compose_two_mappings() {
        // A->B: each pos maps to pos+1
        let ab = CharMapping::new(vec![(0, 1), (1, 2), (2, 3)]);
        // B->C: each pos maps to pos+1
        let bc = CharMapping::new(vec![(0, 1), (1, 2), (2, 3), (3, 4)]);
        let ac = ab.compose(&bc).unwrap();
        // A->C should shift by 2
        assert_eq!(ac.to_normalized(0).unwrap(), 2);
        assert_eq!(ac.to_normalized(1).unwrap(), 3);
        assert_eq!(ac.to_normalized(2).unwrap(), 4);
    }

    #[test]
    fn compose_three_mappings_round_trip() {
        // SUCCESS CRITERION 3: compose 3 mappings, round-trip positions
        let m1 = CharMapping::new(vec![(0, 0), (1, 1), (2, 2), (3, 3), (4, 4)]);
        let m2 = CharMapping::new(vec![(0, 0), (1, 1), (3, 2), (4, 3)]);
        let m3 = CharMapping::new(vec![(0, 1), (1, 2), (2, 3), (3, 4)]);

        let m12 = m1.compose(&m2).unwrap();
        let m123 = m12.compose(&m3).unwrap();

        // Forward mapping works
        let final_0 = m123.to_normalized(0).unwrap();
        let final_1 = m123.to_normalized(1).unwrap();

        // Round-trip: to_original(to_normalized(pos)) == pos
        assert_eq!(m123.to_original(final_0).unwrap(), 0);
        assert_eq!(m123.to_original(final_1).unwrap(), 1);
    }

    #[test]
    fn empty_mapping_returns_error() {
        let m = CharMapping::new(vec![]);
        assert!(m.to_normalized(0).is_err());
        assert!(m.to_original(0).is_err());
    }

    #[test]
    fn accessors() {
        let m = CharMapping::new(vec![(0, 0), (1, 1)]);
        assert_eq!(m.len(), 2);
        assert!(!m.is_empty());
        assert_eq!(m.alignments().len(), 2);

        let empty = CharMapping::new(vec![]);
        assert!(empty.is_empty());
    }
}

#[cfg(test)]
mod proptests {
    use super::*;
    use proptest::prelude::*;

    proptest! {
        #[test]
        fn identity_always_round_trips(len in 1u32..1000, pos in 0u32..999) {
            let len = len.max(1);
            let pos = pos % len;
            let m = CharMapping::identity(len);
            prop_assert_eq!(m.to_normalized(pos).unwrap(), pos);
            prop_assert_eq!(m.to_original(pos).unwrap(), pos);
        }

        #[test]
        fn compose_with_identity_is_noop(
            pairs in proptest::collection::vec((0u32..100, 0u32..100), 1..50)
        ) {
            let mut sorted: Vec<(u32, u32)> = pairs;
            sorted.sort_by_key(|p| p.0);
            sorted.dedup_by_key(|p| p.0);
            if sorted.is_empty() { return Ok(()); }

            let max_norm = sorted.iter().map(|p| p.1).max().unwrap_or(0);
            let m = CharMapping::new(sorted);
            let id = CharMapping::identity(max_norm + 1);
            let composed = m.compose(&id).unwrap();

            // Composed should produce same results as m for all original positions
            for &(orig, _) in m.alignments() {
                prop_assert_eq!(
                    composed.to_normalized(orig).unwrap(),
                    m.to_normalized(orig).unwrap()
                );
            }
        }
    }
}
