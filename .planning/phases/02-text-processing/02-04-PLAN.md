```rust
impl Normalizer for RemovePunctuation {
    fn normalize(&self, input: &str) -> Result<NormalizationResult, NormalizeError> {
        if input.is_empty() { return Err(NormalizeError::EmptyInput); }

        let mut normalized = String::with_capacity(input.len());
        let mut alignments = Vec::new();
        let mut norm_offset: u32 = 0;

        for (byte_offset, ch) in input.char_indices() {
            if !ch.is_ascii_punctuation() {
                alignments.push((byte_offset as u32, norm_offset));
                normalized.push(ch);
                norm_offset += ch.len_utf8() as u32;
            }
            // Removed chars: no alignment entry (1:0 mapping)
        }

        if alignments.is_empty() {
            return Err(NormalizeError::CompositionFailed);
        }

        let mapping = CharMapping::new(alignments)?;
        Ok(NormalizationResult { text: normalized, mapping })
    }

    fn name(&self) -> &str { "punctuation" }
    fn cost(&self) -> f32 { 0.2 }
}
```
