**Success Criterion 1:** Normalize "Hello  WORLD" with Lowercase + Whitespace -> "hello world" with valid CharMapping

```rust
#[test]
fn success_criterion_1_lowercase_whitespace_normalization() {
    let processor = TextProcessor::new(
        vec![
            Box::new(LowercaseNormalizer),
            Box::new(WhitespaceNormalizer::default()),
        ],
        Box::new(WordTokenizer),
        ExecutionMode::All,
    ).unwrap();

    let result = processor.process("Hello  WORLD").unwrap();

    assert_eq!(result.normalized, "hello world");
    assert_eq!(result.tokens.len(), 2);

    let token_texts: Vec<&str> = result.tokens.iter()
        .map(|t| result.text_store.resolve(t.text_id).unwrap())
        .collect();
    assert_eq!(token_texts, vec!["hello", "world"]);

    // Verify CharMapping is valid
    let mapping = result.composed_mapping.as_ref().unwrap();
    assert_eq!(mapping.to_normalized(0).unwrap(), 0); // 'H' -> 'h'
}
```

```rust
#[test]
fn success_criterion_4_full_pipeline_three_normalizers() {
    let processor = TextProcessor::new(
        vec![
            Box::new(LowercaseNormalizer),
            Box::new(WhitespaceNormalizer::default()),
            Box::new(UnicodeNormalizer::nfc()),
        ],
        Box::new(WordTokenizer),
        ExecutionMode::All,
    ).unwrap();

    let input = "  Hello   WORLD   café  ";
    let result = processor.process(input).unwrap();

    assert!(!result.tokens.is_empty(), "Pipeline should produce tokens");

    for token in &result.tokens {
        let start = token.span.start() as usize;
        let end = token.span.end() as usize;
        assert!(start <= end, "Span start {} > end {}", start, end);
        assert!(
            end <= result.normalized.len(),
            "Span end {} exceeds normalized text length {}",
            end, result.normalized.len()
        );
        assert!(result.normalized.is_char_boundary(start));
        assert!(result.normalized.is_char_boundary(end));

        let span_text = &result.normalized[start..end];
        let resolved = result.text_store.resolve(token.text_id).unwrap();
        assert_eq!(span_text, resolved);
    }

    assert_eq!(result.layers.len(), 3);
    assert!(result.composed_mapping.is_some());
}
```

```rust
fn minimal_vs_all_execution_modes() {
    let make_processor = |mode: ExecutionMode| {
        TextProcessor::new(
            vec![
                Box::new(LowercaseNormalizer),
                Box::new(WhitespaceNormalizer::default()),
            ],
            Box::new(WordTokenizer),
            mode,
        ).unwrap()
    };

    let input = "Hello  WORLD";

    assert_eq!(all_result.tokens.len(), min_result.tokens.len());
```
