**Implement CharTokenizer:**

```rust
use unicode_segmentation::UnicodeSegmentation;

impl Tokenizer for CharTokenizer {
    fn tokenize(&self, text: &str, _mapping: &CharMapping, store: &mut TextStoreBuilder) -> Result<Vec<Token>, TokenizeError> {
        if text.is_empty() { return Ok(vec![]); }

        let mut tokens = Vec::new();
        for (byte_offset, grapheme) in text.grapheme_indices(true) {
            let text_id = store.intern(grapheme);
            let span = Span::new(byte_offset as u32, (byte_offset + grapheme.len()) as u32);
            tokens.push(Token::new(text_id, span, TokenKind::Regular));
        }
        Ok(tokens)
    }

    fn name(&self) -> &str { "char" }
}
```

Key: `grapheme_indices(true)` uses Extended Grapheme Clusters (UAX#29).

**Implement WordTokenizer:**

```rust
impl Tokenizer for WordTokenizer {
    fn tokenize(&self, text: &str, _mapping: &CharMapping, store: &mut TextStoreBuilder) -> Result<Vec<Token>, TokenizeError> {
        if text.is_empty() { return Ok(vec![]); }

        let mut tokens = Vec::new();
        let mut word_start: Option<usize> = None;

        for (byte_idx, ch) in text.char_indices() {
            if ch.is_whitespace() {
                if let Some(start) = word_start {
                    let word = &text[start..byte_idx];
                    let id = store.intern(word);
                    tokens.push(Token::new(id, Span::new(start as u32, byte_idx as u32), TokenKind::Regular));
                    word_start = None;
                }
            } else if word_start.is_none() {
                word_start = Some(byte_idx);
            }
        }

        if let Some(start) = word_start {
            let word = &text[start..];
            let id = store.intern(word);
            tokens.push(Token::new(id, Span::new(start as u32, text.len() as u32), TokenKind::Regular));
        }

        Ok(tokens)
    }

    fn name(&self) -> &str { "word" }
}
```

Key: `char::is_whitespace()` covers ASCII + Unicode whitespace. Punctuation NOT split (locked).
