**Update `process/mod.rs`** — replace the stub `TextProcessor` implementation:

```rust
use crate::error::ProcessError;
use crate::normalize::{Normalizer, NormalizationResult};
use crate::tokenize::Tokenizer;
use crate::char_mapping::CharMapping;
use crate::text_store::TextStoreBuilder;
use crate::token::Token;

use std::sync::Arc;

pub mod layer;
pub mod processed_text;

pub use layer::{ExecutionMode, NormalizationLayer};
pub use processed_text::ProcessedText;

pub struct TextProcessor {
    normalizers: Vec<Box<dyn Normalizer>>,
    tokenizer: Box<dyn Tokenizer>,
    mode: ExecutionMode,
}

impl TextProcessor {
    pub fn new(
        normalizers: Vec<Box<dyn Normalizer>>,
        tokenizer: Box<dyn Tokenizer>,
        mode: ExecutionMode,
    ) -> Result<Self, ProcessError> {
        // Validate normalizer name uniqueness
        let mut seen_names = std::collections::HashSet::new();
        for norm in &normalizers {
            if !seen_names.insert(norm.name().to_string()) {
                return Err(ProcessError::Configuration(
                    format!("duplicate normalizer name: '{}'", norm.name())
                ));
            }
        }
        Ok(Self { normalizers, tokenizer, mode })
    }

    pub fn process(&self, input: &str) -> Result<ProcessedText, ProcessError> {
        // Handle empty input at TextProcessor level — do NOT call normalizers
        if input.is_empty() {
            let store = TextStoreBuilder::new();
            return Ok(ProcessedText {
                original: String::new(),
                layers: Vec::new(),
                normalized: String::new(),
                tokens: Vec::new(),
                composed_mapping: None,
                text_store: store.build(),
                skipped_normalizers: Vec::new(),
            });
        }

        let mut current_text = input.to_string();
        let mut layers: Vec<NormalizationLayer> = Vec::new();
        let mut composed_mapping: Option<CharMapping> = None;
        let mut skipped_normalizers: Vec<String> = Vec::new();

        for normalizer in &self.normalizers {
            let result = normalizer.normalize(&current_text)
                .map_err(|source| ProcessError::NormalizationFailed {
                    normalizer: normalizer.name().to_string(),
                    source,
                })?;

            // Compose mapping
            composed_mapping = Some(match &composed_mapping {
                None => result.mapping.clone(),
                Some(prev) => prev.compose(&result.mapping)
                    .map_err(|source| ProcessError::NormalizationFailed {
                        normalizer: normalizer.name().to_string(),
                        source,
                    })?,
            });

            // Store layer based on execution mode
            match self.mode {
                ExecutionMode::All => {
                    layers.push(NormalizationLayer {
                        name: normalizer.name().to_string(),
                        text: result.text.clone(),
                        mapping: result.mapping,
                    });
                }
                ExecutionMode::Minimal => {
                    // In Minimal mode, no layers are stored by default
                    // (future: store only layers required by configured analyzers)
                    skipped_normalizers.push(normalizer.name().to_string());
                }
            }

            current_text = result.text;
        }

        // Build composed mapping for tokenization
        let final_mapping = composed_mapping.clone().unwrap_or_else(|| {
            CharMapping::identity(current_text.len() as u32)
                .unwrap_or_else(|_| CharMapping::identity(1).unwrap())
        });

        // Run tokenizer on final normalized text
        let mut store = TextStoreBuilder::new();
        let tokens = self.tokenizer
            .tokenize(&current_text, &final_mapping, &mut store)
            .map_err(|source| ProcessError::TokenizationFailed {
                tokenizer: self.tokenizer.name().to_string(),
                source,
            })?;

        let text_store = store.build();

        Ok(ProcessedText {
            original: input.to_string(),
            layers,
            normalized: current_text,
            tokens,
            composed_mapping,
            text_store,
            skipped_normalizers,
        })
    }
}
```

```rust
use crate::char_mapping::CharMapping;
use crate::error::ProcessError;
use crate::token::Token;
use crate::text_store::TextStore;
use super::layer::NormalizationLayer;
use std::sync::Arc;

#[derive(Debug)]
pub struct ProcessedText {
    pub original: String,
    pub layers: Vec<NormalizationLayer>,
    pub normalized: String,
    pub tokens: Vec<Token>,
    pub composed_mapping: Option<CharMapping>,
    pub text_store: Arc<TextStore>,
    pub skipped_normalizers: Vec<String>,
}

impl ProcessedText {
    /// Look up a layer by normalizer name.
    pub fn layer(&self, name: &str) -> Option<&NormalizationLayer> {
        self.layers.iter().find(|l| l.name == name)
    }

    /// Compute the composed CharMapping from original text to the output
    /// of the named normalization layer.
    pub fn mapping_to_layer(&self, name: &str) -> Result<CharMapping, ProcessError> {
        let target_idx = self.layers.iter().position(|l| l.name == name)
            .ok_or_else(|| ProcessError::LayerNotFound(name.to_string()))?;

        let mut composed: Option<CharMapping> = None;
        for layer in &self.layers[..=target_idx] {
            composed = Some(match composed {
                None => layer.mapping.clone(),
                Some(prev) => prev.compose(&layer.mapping)
                    .map_err(|source| ProcessError::NormalizationFailed {
                        normalizer: layer.name.clone(),
                        source,
                    })?,
            });
        }

        composed.ok_or_else(|| ProcessError::LayerNotFound(name.to_string()))
    }
}
```
