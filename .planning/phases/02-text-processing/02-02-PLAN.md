**Implement Lowercase normalizer:**

```rust
impl Normalizer for Lowercase {
    fn normalize(&self, input: &str) -> Result<NormalizationResult, NormalizeError> {
        if input.is_empty() {
            return Err(NormalizeError::EmptyInput);
        }

        let mut normalized = String::with_capacity(input.len());
        let mut alignments = Vec::new();
        let mut norm_byte_offset: u32 = 0;

        for (orig_byte_offset, ch) in input.char_indices() {
            for lower_ch in ch.to_lowercase() {
                alignments.push((orig_byte_offset as u32, norm_byte_offset));
                normalized.push(lower_ch);
                norm_byte_offset += lower_ch.len_utf8() as u32;
            }
        }

        let mapping = CharMapping::new(alignments)
            .map_err(|_| NormalizeError::InvalidMapping)?;
        Ok(NormalizationResult { text: normalized, mapping })
    }

    fn name(&self) -> &str { "lowercase" }
    fn cost(&self) -> f32 { 0.1 }
}
```

```rust
impl Normalizer for WhitespaceNormalizer {
    fn normalize(&self, input: &str) -> Result<NormalizationResult, NormalizeError> {
        if input.is_empty() {
            return Err(NormalizeError::EmptyInput);
        }

        let mut normalized = String::with_capacity(input.len());
        let mut alignments = Vec::new();
        let mut norm_byte_offset: u32 = 0;
        let mut in_whitespace = false;
        let mut leading = true;

        for (orig_byte_offset, ch) in input.char_indices() {
            if ch.is_whitespace() {
                if leading { continue; }
                if !in_whitespace {
                    alignments.push((orig_byte_offset as u32, norm_byte_offset));
                    normalized.push(' ');
                    norm_byte_offset += 1;
                    in_whitespace = true;
                }
            } else {
                leading = false;
                in_whitespace = false;
                alignments.push((orig_byte_offset as u32, norm_byte_offset));
                normalized.push(ch);
                norm_byte_offset += ch.len_utf8() as u32;
            }
        }

        // Trim trailing space
        if normalized.ends_with(' ') {
            normalized.pop();
            if let Some(&(_, last_norm)) = alignments.last() {
                if last_norm == norm_byte_offset - 1 {
                    alignments.pop();
                }
            }
        }

        if normalized.is_empty() {
            return Err(NormalizeError::EmptyInput);
        }

        let mapping = CharMapping::new(alignments)
            .map_err(|_| NormalizeError::InvalidMapping)?;
        Ok(NormalizationResult { text: normalized, mapping })
    }

    fn name(&self) -> &str { "whitespace" }
    fn cost(&self) -> f32 { 0.1 }
}
```

Key: Leading whitespace skipped, runs collapsed to single space (only first gets mapping entry), trailing trimmed.
